cmake_minimum_required(VERSION 3.20)
project(adcs_ground_station LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Default to an optimized build when no build type is given (single-config generators).
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(ADCS_BUILD_TESTS "Build unit tests" ON)

# MinGW: link runtimes statically so binaries don't pick up a mismatched
# libstdc++/libgomp DLL from PATH (e.g. another toolchain on PATH).
if(MINGW)
  add_link_options(-static)
endif()

# --- adcs_core: SGP4 and TEME->ECEF (originally generated by MATLAB Coder) ---
file(GLOB ADCS_CORE_SOURCES CONFIGURE_DEPENDS adcs_core/*.cpp)
add_library(adcs_core STATIC ${ADCS_CORE_SOURCES})
target_include_directories(adcs_core PUBLIC adcs_core)

# The core class guards its state with OpenMP nest locks (thread-safe methods).
find_package(OpenMP REQUIRED COMPONENTS CXX)
target_link_libraries(adcs_core PUBLIC OpenMP::OpenMP_CXX)

# --- adcs_contact: adcs::contactState (CelesTrak pull -> SGP4 -> state for the contact uplink) ---
add_library(adcs_contact STATIC src/contact_state.cpp src/tle.cpp src/config.cpp)
target_include_directories(adcs_contact PUBLIC include)
target_link_libraries(adcs_contact PUBLIC adcs_core)
# Repo root, for the default data dir (<repo>/output).
cmake_path(GET CMAKE_CURRENT_SOURCE_DIR PARENT_PATH ADCS_REPO_ROOT)
target_compile_definitions(adcs_contact PRIVATE ADCS_REPO_ROOT="${ADCS_REPO_ROOT}")

# --- contact_state: command-line wrapper for the ground-station software (prints JSON) ---
add_executable(contact_state apps/contact_state.cpp)
target_link_libraries(contact_state PRIVATE adcs_contact)

if(ADCS_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)

  # Style check (Google C++ style, settings in CPPLINT.cfg). Runs with ctest
  # when cpplint is installed (pip install cpplint; included in the dev image).
  find_program(CPPLINT_EXE cpplint)
  if(CPPLINT_EXE)
    add_test(NAME cpplint
             COMMAND ${CPPLINT_EXE} --quiet --recursive include src apps tests
             WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  else()
    message(STATUS "cpplint not found: style check skipped (pip install cpplint)")
  endif()
endif()
