cmake_minimum_required(VERSION 3.20)
project(adcs_ground_station LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Default to an optimized build when no build type is given (single-config generators).
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(ADCS_BUILD_TESTS "Build unit tests" ON)

# MinGW: link runtimes statically so binaries don't pick up a mismatched
# libstdc++/libgomp DLL from PATH (e.g. another toolchain on PATH).
if(MINGW)
  add_link_options(-static)
endif()

# --- adcs_core: SGP4, TEME->ECEF and window search (originally generated by MATLAB Coder) ---
file(GLOB ADCS_CORE_SOURCES CONFIGURE_DEPENDS adcs_core/*.cpp)
add_library(adcs_core STATIC ${ADCS_CORE_SOURCES})
target_include_directories(adcs_core PUBLIC adcs_core)

# The core class guards its state with OpenMP nest locks (thread-safe methods).
find_package(OpenMP REQUIRED COMPONENTS CXX)
target_link_libraries(adcs_core PUBLIC OpenMP::OpenMP_CXX)

# --- adcs_windows: hand-written wrapper (TLE parsing, propagation, pass prediction) ---
add_library(adcs_windows STATIC src/window_predictor.cpp src/config.cpp)
target_include_directories(adcs_windows PUBLIC include)
target_link_libraries(adcs_windows PUBLIC adcs_core)
# Repo root, for the default data dir (<repo>/output).
cmake_path(GET CMAKE_CURRENT_SOURCE_DIR PARENT_PATH ADCS_REPO_ROOT)
target_compile_definitions(adcs_windows PRIVATE ADCS_REPO_ROOT="${ADCS_REPO_ROOT}")

# --- run_demo: 24 h pass prediction for a sample TLE ---
add_executable(run_demo apps/run_demo.cpp)
target_link_libraries(run_demo PRIVATE adcs_windows)

if(ADCS_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
